# Optional dependencies for REAL AI-generated-text detection AND the semantic
# (embedding) plagiarism signal.
#
#   * AI Detection  → language-model perplexity + burstiness (distilgpt2)
#   * Plagiarism    → all-MiniLM-L6-v2 embeddings, reported as a review signal
#
# On servers (no GPU) install torch CPU-FIRST, otherwise pip may pull the huge
# CUDA build (~3 GB) from PyPI:
#
#     pip install torch==2.14.0 --index-url https://download.pytorch.org/whl/cpu
#     pip install -r requirements-ai.txt
#
# On first use the models download automatically to ~/.cache/huggingface
# (distilgpt2 ~330 MB, all-MiniLM-L6-v2 ~90 MB; set HF_HOME to relocate).
# Expect roughly 1.5-2 GB extra RAM once both are loaded. Without these
# dependencies the app still runs: AI Detection falls back to heuristics and
# plagiarism falls back to lexical-only detection.

transformers==5.17.0
